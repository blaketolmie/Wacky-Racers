#include <stdio.h>

#include "usb_serial.h"
#include "pwm.h"
#include "pio.h"
#include "delay.h"
#include "panic.h"
#include "target.h"

#define LEFT_PWM_PIO    MOTOR_LEFT_PWM_PIO
#define LEFT_DIR_PIO    MOTOR_LEFT_DIR_PIO
#define RIGHT_PWM_PIO   MOTOR_RIGHT_PWM_PIO
#define RIGHT_DIR_PIO   MOTOR_RIGHT_DIR_PIO

#define PWM_FREQ_HZ 500

#define DUTY_MIN 0
#define DUTY_MAX 100

static const pwm_cfg_t left_pwm_cfg =
{
    .pio = LEFT_PWM_PIO,
    .period = PWM_PERIOD_DIVISOR(PWM_FREQ_HZ),
    .duty = PWM_DUTY_DIVISOR(PWM_FREQ_HZ, 0),
    .align = PWM_ALIGN_LEFT,
    .polarity = PWM_POLARITY_HIGH,
    .stop_state = PIO_OUTPUT_LOW
};

static const pwm_cfg_t right_pwm_cfg =
{
    .pio = RIGHT_PWM_PIO,
    .period = PWM_PERIOD_DIVISOR(PWM_FREQ_HZ),
    .duty = PWM_DUTY_DIVISOR(PWM_FREQ_HZ, 0),
    .align = PWM_ALIGN_LEFT,
    .polarity = PWM_POLARITY_HIGH,
    .stop_state = PIO_OUTPUT_LOW
};

static int clamp(int value, int min, int max)
{
    if (value > max)
        return max;

    if (value < min)
        return min;

    return value;
}

static int abs_int(int value)
{
    if (value < 0)
        return -value;

    return value;
}

/*
   EXACT same direction logic as your working code:

   Forward:
       LEFT_DIR  LOW
       RIGHT_DIR LOW

   Reverse:
       LEFT_DIR  HIGH
       RIGHT_DIR HIGH

   Stop:
       LEFT_DIR  HIGH
       RIGHT_DIR LOW
*/

static void motors_forward(pwm_t left_pwm, pwm_t right_pwm,
                           int left_duty, int right_duty)
{
    left_duty = clamp(left_duty, DUTY_MIN, DUTY_MAX);
    right_duty = clamp(right_duty, DUTY_MIN, DUTY_MAX);

    /* Forward */
    pio_output_low(STAT3_PIO);
    pio_output_high(STAT0_PIO);

    pio_output_high(LEFT_DIR_PIO);
    pio_output_low(RIGHT_DIR_PIO);

    pwm_duty_set(left_pwm, PWM_DUTY_DIVISOR(PWM_FREQ_HZ, left_duty));
    pwm_duty_set(right_pwm, PWM_DUTY_DIVISOR(PWM_FREQ_HZ, right_duty));
}

static void motors_reverse(pwm_t left_pwm, pwm_t right_pwm,
                           int left_duty, int right_duty)
{
    left_duty = clamp(left_duty, DUTY_MIN, DUTY_MAX);
    right_duty = clamp(right_duty, DUTY_MIN, DUTY_MAX);

    /* Reverse */
    pio_output_low(STAT3_PIO);
    pio_output_high(STAT0_PIO);

    pio_output_high(LEFT_DIR_PIO);
    pio_output_high(RIGHT_DIR_PIO);

    pwm_duty_set(left_pwm, PWM_DUTY_DIVISOR(PWM_FREQ_HZ, left_duty));
    pwm_duty_set(right_pwm, PWM_DUTY_DIVISOR(PWM_FREQ_HZ, right_duty));
}

static void motors_stop(pwm_t left_pwm, pwm_t right_pwm)
{
    /* Stop */
    pio_output_high(STAT3_PIO);
    pio_output_low(STAT0_PIO);

    pio_output_high(LEFT_DIR_PIO);
    pio_output_low(RIGHT_DIR_PIO);

    pwm_duty_set(left_pwm, PWM_DUTY_DIVISOR(PWM_FREQ_HZ, 0));
    pwm_duty_set(right_pwm, PWM_DUTY_DIVISOR(PWM_FREQ_HZ, 0));
}

/*
   speed:
       +100 = full forward
       0    = stop
       -100 = full reverse

   steer:
       +100 = left motor faster, right motor slower
       -100 = right motor faster, left motor slower

   This gives steering by changing left/right duty cycle,
   while keeping your proven direction pin logic unchanged.
*/
static void drive_set(pwm_t left_pwm, pwm_t right_pwm, int speed, int steer)
{
    int base_duty;
    int turn;
    int left_duty;
    int right_duty;

    speed = clamp(speed, -100, 100);
    steer = clamp(steer, -100, 100);

    if (speed == 0)
    {
        motors_stop(left_pwm, right_pwm);
        printf("Stopped\r\n");
        fflush(stdout);
        return;
    }

    base_duty = abs_int(speed);
    turn = abs_int(steer);

    left_duty = base_duty;
    right_duty = base_duty;

    if (steer > 0)
    {
        left_duty = base_duty;
        right_duty = base_duty * (100 - turn) / 100;
    }
    else if (steer < 0)
    {
        left_duty = base_duty * (100 - turn) / 100;
        right_duty = base_duty;
    }

    if (speed > 0)
    {
        motors_forward(left_pwm, right_pwm, left_duty, right_duty);
        printf("Forward: left duty = %d, right duty = %d\r\n",
               left_duty, right_duty);
    }
    else
    {
        motors_reverse(left_pwm, right_pwm, left_duty, right_duty);
        printf("Reverse: left duty = %d, right duty = %d\r\n",
               left_duty, right_duty);
    }

    fflush(stdout);
}

static void prompt_command(void)
{
    printf("\r\nEnter speed steer (-100 to 100): ");
    fflush(stdout);
}

static void process_command(pwm_t left_pwm, pwm_t right_pwm)
{
    char buffer[80];
    int speed;
    int steer;

    if (!fgets(buffer, sizeof(buffer), stdin))
        return;

    if (sscanf(buffer, "%d %d", &speed, &steer) == 2)
    {
        drive_set(left_pwm, right_pwm, speed, steer);
        pio_output_toggle(LED_STATUS_PIO);
    }
    else
    {
        printf("Invalid input. Example: 50 25\r\n");
        fflush(stdout);
    }

    prompt_command();
}

int main(void)
{
    pwm_t left_pwm;
    pwm_t right_pwm;

    pio_config_set(LED_STATUS_PIO, PIO_OUTPUT_LOW);
    pio_output_set(LED_STATUS_PIO, LED_ACTIVE);

    pio_config_set(LED_ERROR_PIO, PIO_OUTPUT_LOW);
    pio_output_set(LED_ERROR_PIO, !LED_ACTIVE);

    pio_config_set(STAT0_PIO, PIO_OUTPUT_HIGH);
    pio_config_set(STAT1_PIO, PIO_OUTPUT_HIGH);
    pio_config_set(STAT2_PIO, PIO_OUTPUT_HIGH);
    pio_config_set(STAT3_PIO, PIO_OUTPUT_HIGH);

    pio_config_set(HBRIDGE_ENABLE_PIO, PIO_OUTPUT_HIGH);
    pio_output_high(HBRIDGE_ENABLE_PIO);

    pio_config_set(SLEEP_PIO, PIO_OUTPUT_HIGH);
    pio_output_high(SLEEP_PIO);

    pio_config_set(LEFT_DIR_PIO, PIO_OUTPUT_LOW);
    pio_config_set(RIGHT_DIR_PIO, PIO_OUTPUT_LOW);

    left_pwm = pwm_init(&left_pwm_cfg);
    if (!left_pwm)
        panic(LED_ERROR_PIO, 1);

    right_pwm = pwm_init(&right_pwm_cfg);
    if (!right_pwm)
        panic(LED_ERROR_PIO, 2);

    pwm_channels_start(pwm_channel_mask(left_pwm) | pwm_channel_mask(right_pwm));

    usb_serial_stdio_init();

    motors_stop(left_pwm, right_pwm);

    printf("\r\nWacky Racer USB PWM motor control ready.\r\n");
    printf("Input format: speed steer\r\n");
    printf("Examples:\r\n");
    printf("  50 0    = forward straight\r\n");
    printf(" -50 0    = reverse straight\r\n");
    printf("  50 50   = forward turning one way\r\n");
    printf("  50 -50  = forward turning the other way\r\n");
    printf("  0 0     = stop\r\n");
    fflush(stdout);

    prompt_command();

    while (1)
    {
        process_command(left_pwm, right_pwm);
    }

    return 0;
}
